cmake_minimum_required(VERSION 3.15)
project(obf_php_strings CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(obf_php_strings SHARED src/PhpObf.cpp)
target_include_directories(obf_php_strings PRIVATE "${CMAKE_SOURCE_DIR}/include")
if(MSVC)
    target_compile_options(obf_php_strings PRIVATE /W4 /wd5220 /wd4239 /wd9025)
    target_compile_definitions(obf_php_strings PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
set_target_properties(obf_php_strings PROPERTIES
    PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin/plugins"
    OUTPUT_NAME "obf_php_strings")
