cmake_minimum_required(VERSION 3.15)
project(obf_opaque CXX)

set(CMAKE_CXX_STANDARD 17)

add_library(obf_opaque SHARED
    src/OpaqueObf.cpp
)

target_include_directories(obf_opaque PRIVATE
    "${CMAKE_SOURCE_DIR}/include"
)

set_target_properties(obf_opaque PROPERTIES
    PREFIX ""
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin/plugins"
    OUTPUT_NAME "obf_opaque"
)
