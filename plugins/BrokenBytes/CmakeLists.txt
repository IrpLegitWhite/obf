cmake_minimum_required(VERSION 3.15)
project(obf_broken_bytes CXX)

set(CMAKE_CXX_STANDARD 17)

add_library(obf_broken_bytes SHARED
    src/BrokenBytesObf.cpp
)

target_include_directories(obf_broken_bytes PRIVATE
    "${CMAKE_SOURCE_DIR}/include"
)

if(MSVC)
    target_compile_options(obf_broken_bytes PRIVATE /W4 /wd5220 /wd4239 /wd4828)
    target_compile_definitions(obf_broken_bytes PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()

set_target_properties(obf_broken_bytes PROPERTIES
    PREFIX ""
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin/plugins"
    OUTPUT_NAME "obf_broken_bytes"
)
