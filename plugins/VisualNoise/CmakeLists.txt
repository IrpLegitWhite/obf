cmake_minimum_required(VERSION 3.15)
project(obf_visual_noise CXX)

set(CMAKE_CXX_STANDARD 17)

add_library(obf_visual_noise SHARED
    src/VisualNoiseObf.cpp
)

target_include_directories(obf_visual_noise PRIVATE
    "${CMAKE_SOURCE_DIR}/include"
)

if(MSVC)
    target_compile_options(obf_visual_noise PRIVATE /W4 /wd5220 /wd4239)
    target_compile_definitions(obf_visual_noise PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()

set_target_properties(obf_visual_noise PROPERTIES
    PREFIX ""
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin/plugins"
    OUTPUT_NAME "obf_visual_noise"
)
