cmake_minimum_required(VERSION 3.15)
project(obf_java_mba CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(obf_java_mba SHARED src/JavaMbaObf.cpp)
target_include_directories(obf_java_mba PRIVATE "${CMAKE_SOURCE_DIR}/include")
if(MSVC)
    target_compile_options(obf_java_mba PRIVATE /W4 /wd5220 /wd4239 /wd9025)
    target_compile_definitions(obf_java_mba PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
set_target_properties(obf_java_mba PROPERTIES
    PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin/plugins"
    OUTPUT_NAME "obf_java_mba")
